def write_csv(path: Path, rows: list[dict[str, object]]) -> None:
    with path.open("w", newline="", encoding="utf-8") as fh:
        writer = csv.DictWriter(fh, fieldnames=list(rows[0].keys()))
        writer.writeheader()
   
     writer.writerows(rows)
def load_csvs_to_sqlite(csv_dir: Path, db_path: Path) -> dict[str, int]:
    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(db_path, isolation_level=None)
    conn.execute("PRAGMA journal_mode = W
AL;")
    conn.execute("PRAGMA synchronous = NORMAL;")
    conn.execute("PRAGMA busy_timeout = 5000;")
    conn.execute("PRAGMA foreign_keys = ON;")
    conn.executescript(DDL_DOMAIN)
    conn.execute("PRAGMA foreign_keys = ON;")
    mapping = [
        (
"weather_traffic.csv", "dom_weather_traffic"),
        ("warehouse_stocks.csv", "dom_warehouse_stocks"),
        ("transit_telemetry.csv", "dom_transit_telemetry"),
        ("supplier_sla.csv", "dom_supplier_sla"),
    ]
    loaded: dict[str, int] = {}
   
 conn.execute("BEGIN;")
    try:
        for fname, table in mapping:
            with (csv_dir / fname).open("r", encoding="utf-8") as fh:
                rows = list(csv.DictReader(fh))
            if not rows:
                loaded[table] = 0
         
       continue
            cols = list(rows[0].keys())
            placeholders = ", ".join("?" for _ in cols)
            sql = f"INSERT OR REPLACE INTO {table} ({', '.join(cols)}) VALUES ({placeholders})"
            conn.executemany(sql, [[r[c] for c i
n cols] for r in rows])
            loaded[table] = len(rows)
        conn.execute("COMMIT;")
    except Exception:
        conn.execute("ROLLBACK;")
        raise
    finally:
        conn.close()
    return loaded

def generate_random_batch(
    count: int = 15,
    seed: int = 42,
    conflict_bias: float = 0.65,
) -> list[S
cenarioTask]:
    rng = random.Random(seed)
    users = ("analyst_1", "analyst_2", "manager_1")
    hot_date = DEMO_DATES[len(DEMO_DATES) // 2]  # 2026-11-20 — пиковая дата коллизий
    batch: list[ScenarioTask] = []
    for _ in range(count):
        proc
 = rng.choice(PROCESS_IDS)
        target_date = hot_date if rng.random() < conflict_bias else rng.choice(DEMO_DATES)
        is_auto = rng.random() < 0.25
        trigger = "AUTO" if is_auto else "MANUAL"
        user = "scheduler_bot" if is_auto else rng
.choice(users)
        prio = 1 if is_auto else rng.choice((2, 2, 3))
        delay = round(rng.uniform(0.0, 0.25), 3)
        batch.append(ScenarioTask(proc, target_date, trigger, prio, delay, user))
    return batch

Индексы, представления (VIEW) и критичная деталь инициализации SQLite
Среди 10 индексов схемы ключевыми являются составной индекс очереди idx_task_runs_status_prio(status, effective_priority, queued_at) и 
частичный индекс идемпотентности

 ускоряющий поиск активных дублей. Три SQL-представления выносят типовые аналитические запросы из Python-кода: v_gantt_timeline подготавливает данные для диаграммы Ганта, v_before_after_comparison рассчитывает долю ошибок и среднее время ожидания по каждом
у прогону, а v_conflict_matrix динамически выводит все 24 ресурсных пересечения из таблицы правил.
Важный технический нюанс sqlite3 в Python:
 как отмечается в обсуждениях разработчиков Python, при открытии соединения с autocommit=False транзакция открывает
ся автоматически, а внутри активной транзакции команда PRAGMA foreign_keys = ON 
молча игнорируется SQLite без ошибки
. Для гарантированного включения внешних ключей соединение следует открывать с isolation_level=None, последовательно выполнять PRAGMA (journ
al_mode = WAL, synchronous = NORMAL, busy_timeout = 5000, foreign_keys = ON), повторно подтверждать PRAGMA foreign_keys = ON после executescript() и управлять транзакциями явно через BEGIN/COMMIT.

Упрощение и оптимизация стека разработки
Выбранная в ТЗ монолитная Python-архитектура на базе NiceGUI + asyncio + SQLite уже хорошо подходит для сжатых сроков: отказ от отдельного React-фронтенда, Redis и Docker Compose экономит время на стыковке, а отказ от Streamlit технически оправдан его мод

2. Заменить aiosqlite на встроенный sqlite3 (одно WAL-соединение)
Как отмечается в инженерных разборах и документации библиоте
к, aiosqlite не реализует нативный асинхронный I/O на уровне ОС: он открывает соединение в отдельном рабочем потоке и передаёт запросы через Future, при этом сам движок SQLite всё равно допускает 
строго одного писателя одновременно. Дополнительный практиче
ский риск на демо — известная проблема зависания рабочего потока aiosqlite при завершении приложения или Ctrl+C (в частности, в заметках к релизам Apache Airflow описан отдельный фикс защиты от зависающего потока в aiosqlite 0.22+, а сторонние обёртки выну
ждены принудительно демонизировать его поток).
Локальный синтетический микробенчмарк стандартного sqlite3 (SQLite 3.46.1) с одним постоянным соединением в режиме WAL + synchronous = NORMAL (500 операций) показал среднее время одиночного INSERT 
0,097 мс
 и в
ыборки 50 строк из v_gantt_timeline 
0,181 мс
. При характерной нагрузке MVP (порядка 10–30 записей в секунду на пике стресс-сценария) суммарная задержка в цикле событий не превышает нескольких миллисекунд, что на порядки ниже стандартного 3-секундного тайма
ута сборки страницы в NiceGUI. Важная оговорка для масштабирования: синхронный вызов оправдан именно для локального SQLite при небольших таблицах MVP; при переходе на сетевой PostgreSQL или росте базы до сотен тысяч строк класс Storage следует перевести на
 асинхронный драйвер либо выносить тяжёлые запросы через asyncio.to_thread().
3. Обезопасить многопользовательский UI через nicegui.Event и браузерный режим
Поскольку NiceGUI выполняется в одном процессе Python, переменные уровня модуля разделяются между в
семи подключёнными пользователями, а функция @ui.page('/') запускается заново для каждой новой вкладки. Чтобы исключить утечку ссылок при закрытии вкладок, вместо ручного списка коллбэков в Orchestrator стоит либо использовать встроенный класс nicegui.Even
t() (чей метод subscribe() автоматически регистрирует отписку через client.on_delete), либо явно привязывать ui.context.client.on_delete(lambda: orchestrator.unsubscribe(cb)).
Кроме того, для защиты проекта рекомендуется отказаться от ui.run(native=True) в
 пользу обычного браузерного запуска (localhost:8080): по официальной документации NiceGUI, нативный режим опирается на pywebview и требует современного системного браузерного движка (.NET Framework + EdgeChromium на Windows, актуальный WebKitGTK/Qt на Lin
ux), что создаёт лишнюю точку отказа на чужом ноутбуке. В браузерном режиме многопользовательский сценарий наглядно показывается в двух окнах рядом (analyst_1 и analyst_2), а при необходимости быстрого удалённого показа доступен встроенный техпревью-туннел
ь ui.run(on_air=True) (исключительно для временного демо, не для production).
